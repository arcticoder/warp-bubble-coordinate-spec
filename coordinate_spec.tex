\section{Coordinate Chart}
We adopt a standard spherical coordinate system on the background spacetime:
\begin{align*}
  &\mathrm{Coordinates:}\quad (t,\,r,\,\theta,\,\phi) \\
  &\mathrm{Domains:}\quad
    t \in (-\infty, +\infty),\quad
    r \in [0, +\infty),\quad
    \theta \in [0,\pi],\quad
    \phi \in [0,2\pi).
\end{align*}

\section{Symmetry Assumptions}
\begin{enumerate}
  \item \textbf{Axial symmetry} about the $z$-axis: no $\phi$–dependence.
  \item \textbf{Equatorial reflection symmetry}: invariance under $\theta\!\mapsto\!\pi-\theta$.
  \item \textbf{Compact support in $r$}: warp profiles $A,B,C,D$ nontrivial only for $r\le R$.
\end{enumerate}

\section{Reduced Metric Ansatz}
Imposing the above, the line element becomes
\[
  ds^2 = -A(r)\,dt^2 + B(r)\,dr^2
      + C(r)\,r^2\,d\theta^2
        + D(r)\,r^2\sin^2\theta\,d\phi^2,
\]
with
\[
  A(r),B(r),C(r),D(r) = 
  \begin{cases}
    1, & r>R,\\
    \text{warp–bubble profiles}, & r\le R,
  \end{cases}
\]
and boundary conditions $A(R)=B(R)=C(R)=D(R)=1$.

\section{Worked Example (Sketch)}
\begin{itemize}
  \item Eliminate cross‐terms by staticity \& symmetry.
  \item Enforce $\theta$–independence $\Rightarrow$ functions of $r$ only.
  \item Impose $g_{\mu\nu}=\eta_{\mu\nu}$ for $r>R$.
  \item Rename $g_{tt}=-A(r)$, $g_{rr}=B(r)$, etc.
\end{itemize}

\section{Automation Scripts}
To generate, test and document metric ansätze for any warp–bubble profile, include a `scripts/` folder with:

\begin{verbatim}
warp-bubble-coordinate-spec/
|-- coordinate_spec.tex
|-- scripts/
|   |-- fetch_shape.py
|   |-- generate_ansatz.py
|   +-- requirements.txt
+-- metrics/
    +-- <generated .tex snippets>
\end{verbatim}

\subsection*{1. \texttt{fetch_shape.py}}
\begin{itemize}
  \item \emph{Purpose:} download shape‐profile data from
    \url{https://arcticoder.github.io/warp-bubble-shape-catalog/} and convert to JSON.
  \item \emph{Data Sources:}
    \begin{itemize}
      \item Primary: NPZ format (higher precision) from\\
        \verb|https://github.com/arcticoder/warp-bubble-shape-catalog/raw/refs/heads/main/data/profiles.npz|
      \item Fallback: CSV format from\\
        \verb|https://raw.githubusercontent.com/arcticoder/warp-bubble-shape-catalog/refs/heads/main/data/profile_data.csv|
    \end{itemize}
  \item \emph{Usage:}
    \begin{verbatim}
    python scripts/fetch_shape.py --shape alcubierre
    python scripts/fetch_shape.py --shape natario --format csv
    \end{verbatim}
  \item \emph{Behavior:}
    \begin{enumerate}
      \item Perform HTTP GET on the NPZ data (or CSV if specified).
      \item Parse the data and extract the requested shape profile along with r values.
      \item Convert to JSON format with metadata and save to \verb|scripts/shapes/{shape}.json|.
    \end{enumerate}
\end{itemize}

\subsection*{2. \texttt{generate_ansatz.py}}
\begin{itemize}
  \item \emph{Purpose:} produce a LaTeX snippet defining
    $A(r),B(r),C(r),D(r)$ from the fetched shape.
  \item \emph{Usage:}
    \begin{verbatim}
    python scripts/generate_ansatz.py \
      --shape alcubierre \
      --template ../coordinate_spec.tex \
      --out metrics/alcubierre_ansatz.tex
    \end{verbatim}
  \item \emph{Behavior:}
    \begin{enumerate}
      \item Load \verb|scripts/shapes/{shape}.json|.
      \item Use Jinja2 (or simple string templates) to substitute:
        \[
          A(r)=1 - f(r),\quad B(r)=\frac{1}{1 - f(r)},\;\dots
        \]
      \item Emit a standalone TeX fragment in \verb|metrics/|.
    \end{enumerate}
\end{itemize}

\subsection*{Dependencies \& Installation}
Add a \texttt{scripts/requirements.txt}:
\begin{verbatim}
requests>=2.25
jinja2>=3.0
sympy>=1.8      # if you extend to symbolic curvature computation
numpy>=1.20     # for handling NPZ data format
\end{verbatim}
Install with:
\begin{verbatim}
pip install -r scripts/requirements.txt
\end{verbatim}

\section{Testing \& Validation}
To verify the automation pipeline works correctly:

\begin{enumerate}
  \item Test data fetching:
    \begin{verbatim}
    python scripts/fetch_shape.py --shape alcubierre
    \end{verbatim}
    This should create \verb|scripts/shapes/alcubierre.json| with shape profile data.
    
  \item Test ansatz generation:
    \begin{verbatim}
    python scripts/generate_ansatz.py \
      --shape alcubierre \
      --out metrics/alcubierre_test.tex
    \end{verbatim}
    This should create a LaTeX snippet defining the metric components.
    
  \item Verify the output contains proper mathematical expressions:
    \[
      A(r) = 1 - (\text{interpolated\_alcubierre}(r)), \quad
      B(r) = 1 / (1 - (\text{interpolated\_alcubierre}(r)))
    \]
\end{enumerate}

The scripts automatically handle data format conversion, error checking, and fallback between NPZ and CSV sources. All generated content includes proper LaTeX formatting for mathematical expressions.
